%------------------------------------- OBJECTIVE FUNCTION ------------------------------------

\subsubsection{Objective Function}

\begin{align}
\textbf{\textit{minimize}} \quad 
&
\sum_{p \in \mathcal{P}} \sum_{s \in \mathcal{S}}
\Big(
zco_{p,s} \, PenConfirmedOrders_p
\nonumber \\
& + zsp_{p,s} \, PenPredictiveDemand_p
\Big)
\nonumber \\
&+
\sum_{(p,b,s) \in PBS^{supply}}
\Big(
zsco_{p,b,s} \, PenConfirmedOrders_p
+ zsd_{p,b,s} \, PenPredictiveDemand_p
+ zrp_{p,b,s} \, PenObjectiveStock_p
\Big)
\nonumber \\
&+
\sum_{p \in \mathcal{P}} zso_{p} \, PenObjectiveStock_p \cdot WeightServiceLevel
\nonumber \\
&+
\sum_{p \in \mathcal{P}} \sum_{s \in \mathcal{S}} \Big( WACC \cdot UnitCost_{P} \cdot \frac{ss_{p,w-1}+ss_{p,w}}{2} \Big)
\cdot WeightStock
\nonumber \\
&+
\sum_{m \in \mathcal{M}} \sum_{s \in \mathcal{S}}
ST_{m,s} \, CostSetup_m \cdot WeightSetup
\nonumber \\
&+
\sum_{m \in \mathcal{M}^{ghost}} \sum_{s \in \mathcal{S}}
\sum_{\substack{p \in \mathcal{P},\ r \in \mathcal{R}_p \cap \mathcal{R}^{ghost},\ o \in \mathcal{O}(p,r):\\ m(p,r,o)=m}}
OutsourcingCost_m \, x_{p,r,o,s}
\label{eq:objective_function}
\end{align}

The outsourcing cost is incurred at the production (order placement) shift $s$, independent of the lead time $L$ before the material is received into stock.

\subsubsection{Constraints}

\noindent\text{---------------------------------------------------------------------------------------------------------}

\subsubsection*{A. Inventory, Consumption, and Routing Flow}

To capture the outsourcing lead time, define the quantity of product $p$ \emph{made available into stock} in shift $s$ as the last-operation output of in-house routings produced in shift $s$ plus the last-operation output of ghost routings produced $L$ shifts earlier:
\begin{align}
AvailProd_{p,s}
=
\sum_{\substack{r \in \mathcal{R}_p \\ r \notin \mathcal{R}^{ghost}}}
x_{p,r,o^{last}_{p,r},s}
\;+\;
\sum_{\substack{r \in \mathcal{R}_p \\ r \in \mathcal{R}^{ghost}}}
x_{p,r,o^{last}_{p,r},\,s-L}
\label{eq:avail_prod}
\end{align}
where ghost terms with $s-L < 1$ are omitted (production placed before the horizon start cannot be received within it). All stock balances below use $AvailProd_{p,s}$ in place of the plain sum $\sum_{r \in \mathcal{R}_p} x_{p,r,o^{last}_{p,r},s}$.

\noindent\textbf{A1. Stock balance of coil products and other purchased components}
\begin{align}
ss_{p,s}
&=
ss_{p,s-1}
- \sum_{p_1 \in \mathcal{P}} y_{p_1,p,s}
+ StockArrivals_{p,s}
\label{eq:A1_coil_balance}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{coil} \, \cup \, \mathcal{P}_{other}, \quad s = 2,\dots,45$
\end{center}

\noindent\textbf{A2. Stock balance of coil products and other purchased components, initial shift}
\begin{align}
ss_{p,1}
&=
StockStart_p
- \sum_{p_1 \in \mathcal{P}} y_{p_1,p,1}
+ StockArrivals_{p,1}
\label{eq:A2_coil_initial}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{coil} \, \cup \, \mathcal{P}_{other}$
\end{center}

\noindent\textbf{A3. Stock balance of intermediate products}
\begin{align}
ss_{p,s}
&=
ss_{p,s-1}
- \sum_{p_1 \in \mathcal{P}} y_{p_1,p,s}
+ AvailProd_{p,s}
\label{eq:A3_intermediate_balance}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{intermediate}, \quad s = 2,\dots,45$
\end{center}

\noindent\textbf{A4. Stock balance of intermediate products, initial shift}
\begin{align}
ss_{p,1}
&=
StockStart_p
- \sum_{p_1 \in \mathcal{P}} y_{p_1,p,1}
+ AvailProd_{p,1}
\label{eq:A4_intermediate_initial}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{intermediate}$
\end{center}


\noindent\textbf{A5. Stock balance of finished products}
\begin{align}
ss_{p,s}
&=
ss_{p,s-1}
- D_{p,s}
+ AvailProd_{p,s}
+ zsp_{p,s}
- zsp_{p,s-1}
\label{eq:A5_finished_balance}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{finished}, \quad s = 2,\dots,45$
\end{center}

\noindent\textbf{A6. Stock balance of finished products, initial shift}
\begin{align}
ss_{p,1}
&=
StockStart_p
- D_{p,1}
+ AvailProd_{p,1}
+ zsp_{p,1}
\label{eq:A6_finished_initial}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{finished}$
\end{center}


\noindent\textbf{A5b. Stock balance of boundary products (Cajamar)}

Boundary products carry no Cajamar demand $D_{p,s}$; instead they are dispatched to their destination branch through the transfer variable $tr$. Only transfers dispatched in the current shift drain Cajamar stock, and $tr_{p,b,s}=0$ whenever $s \notin \mathcal{S}^{tr}$.
\begin{align}
ss_{p,s}
&=
ss_{p,s-1}
+ AvailProd_{p,s}
- \sum_{p_1 \in \mathcal{P}} y_{p_1,p,s}
- \sum_{\substack{b \in \mathcal{B}^{dest}:\\ p \in \mathcal{P}^{supply}_b}} tr_{p,b,s}
\label{eq:A9b_boundary_balance}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{boundary}, \quad s = 2,\dots,45$
\end{center}

\noindent\textbf{A6b. Stock balance of boundary products (Cajamar), initial shift}
\begin{align}
ss_{p,1}
&=
StockStart_p
+ AvailProd_{p,1}
- \sum_{p_1 \in \mathcal{P}} y_{p_1,p,1}
- \sum_{\substack{b \in \mathcal{B}^{dest}:\\ p \in \mathcal{P}^{supply}_b}} tr_{p,b,1}
\label{eq:A10b_boundary_initial}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}_{boundary}$
\end{center}



\noindent\textbf{A7. WIP balance between routing operations}
\begin{align}
wip_{p,r,o,s}
&=
wip_{p,r,o,s-1}
+ x_{p,r,o,s}
- x_{p,r,succ(p,r,o),s}
\label{eq:A7_wip_balance}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}, \quad
\forall r \in \mathcal{R}_p, \quad
\forall o \in \mathcal{O}^{int}(p,r), \quad
s = 2,\dots,45$
\end{center}

\noindent\textbf{A8. WIP balance between routing operations, initial shift}
\begin{align}
wip_{p,r,o,1}
&=
WIPStart_{p,r,o}
+ x_{p,r,o,1}
- x_{p,r,succ(p,r,o),1}
\label{eq:A8_wip_initial}
\end{align}
\begin{center}
$\forall p \in \mathcal{P}, \quad
\forall r \in \mathcal{R}_p, \quad
\forall o \in \mathcal{O}^{int}(p,r)$
\end{center}

Because $wip_{p,r,o,s} \ge 0$, constraints \eqref{eq:A7_wip_balance} and \eqref{eq:A8_wip_initial} ensure that operation $succ(p,r,o)$ cannot process more than the quantity already available from previous WIP plus the quantity processed in operation $o$ in the same shift.

\noindent\textbf{A9. Component consumption linked to first-operation production start}

\begin{itemize}

\item The model is a shift-bucketed planning model. It decides production quantities by product, routing, operation, and shift, but does not sequence products or operations within each shift.

\item Each operation belongs to one routing and is assigned to one machine. Machine capacity is checked at aggregate shift level using $UnitProductionTime_{p,r,o}$.

\item Routing flow is represented through operation-indexed WIP variables. Operation $o+1$ may be processed in the same shift as operation $o$, but no explicit intra-shift sequence is modeled.

\item Each product active in a shift is assigned to exactly one routing, enforced directly through the first-operation activity flags $k_{p,r,o^{first}_{p,r},s}$. At most one routing's first operation may be active per product per shift. Once quantity enters a routing, it cannot switch routing because WIP is indexed by product and routing.

\item Minimum lot size is enforced locally at the first operation of each routing. A lot is defined as either a single active shift or an adjacent pair of shifts connected by a continuation. If the first operation is active in shift $s$ with no continuation from $s-1$, its production alone must meet $MinLotSize_p$. If it continues from $s-1$ to $s$, the combined production of the two shifts must meet $MinLotSize_p$. No campaign memory beyond one shift is maintained; three or more consecutive shifts are evaluated only as overlapping adjacent pairs.

\item Stockable intermediate products are represented as products in $\mathcal{P}_{intermediate}$ and have inventory balance constraints. Non-stockable intermediate states inside a routing are represented only through $wip_{p,r,o,s}$.

\item Component consumption is assumed to occur when the first operation of the selected routing is processed. Component choices are made at product-shift level, and each component choice group is mandatory whenever the product starts production.

\item Components consumed in a shift must be available at the start of that shift. Same-shift arrivals and same-shift production of stockable intermediate products are not available for component consumption unless explicitly included in the stock-precedence constraints.

\item Demand $D_{p,s}$ includes confirmed orders and predictive demand. The slack $zsp_{p,s}$ represents cumulative unmet total demand, while $zco_{p,s}$ measures priority-service violation for the confirmed-order subset of demand.

\item Objective stock targets are modeled as soft constraints through penalized shortage variables and may be violated when capacity or material availability is insufficient.

\item Setup times are approximated from the set of distinct setup characteristics active in each machine-shift bucket. The formulation does not sequence setup transitions within a shift.

\item Setup carryover is represented through a single setup reference operation per machine and shift boundary. If an operation is continued from shift $s-1$ to shift $s$, that operation is forced to be the setup reference operation. If no operation is continued, the setup reference chooses one operation from the previous shift to be the last operation processed on the machine.

\item All regular, ink, and varnish carryover variables are derived from the same setup reference operation, ensuring that the carried setup characteristics are mutually consistent.

\item Ink setup is computed by ink position, and the total ink setup time of a machine-shift is approximated as the maximum setup time across ink positions.

\item Machine availability is constrained at shift level through $MachineAvailableHours_{m,s}$. This parameter captures the number of production hours that machine $m$ may operate in shift $s$, including planned shutdowns, reduced availability, or maintenance windows.

\item Crew availability is modeled at area level through $Crew_{a,s}$. Each machine belongs to exactly one production area, represented by a fixed mapping parameter $\mu_{m,a} \in \{0,1\}$. The total number of active machines in each area is limited by the number of available crews in that area and shift.

\item Each active machine consumes a fixed crew requirement, optionally machine-specific through $CrewConsumption_m$. If all machines consume one crew, then $CrewConsumption_m=1$ for all $m \in \mathcal{M}$.

\item Demand, processing times, setup coefficients, arrivals, stock targets, and consumption factors are deterministic inputs.

\item Subcontracting is modeled through \emph{ghost machines} $\mathcal{M}^{ghost}$, disjoint from the in-house machines $\mathcal{M}$. Each subcontracted product is given a single-operation ghost routing $r \in \mathcal{R}^{ghost}$ on a ghost machine, whose only operation is simultaneously the first and last operation, so its output feeds the product stock balance like a one-step in-house routing. Ghost machines carry no processing time, setup, lot-sizing, continuation, or crew logic, and are therefore excluded from in-house routing exclusivity (A13), minimum-lot (A16), continuation (A17), setup (A18--A19, D) and machine-time / crew (B1, B3--B5) constraints; their only capacity limit is the per-shift quantity cap B\_GHOST. They do, however, keep the activation binary $k$ (A15) and consume BOM components like an in-house routing (A9--A12): we supply the input materials to the subcontractor, who performs the operation.

\item The ghost per-shift capacity $GhostCapacity_{m,s}$ is obtained from the weekly subcontracting quote divided by the number of shifts in a full business week ($5 \times 3 = 15$). Applying this per-shift cap to every modeled shift prorates a partial first week linearly (a week with fewer modeled shifts receives a proportionally smaller effective weekly capacity) and prevents the solver from front-loading a whole week's subcontracting into a single shift. This is an approximation; it does not yet discount volume already firmed with the subcontractor.

\item Subcontracted material is subject to an outsourcing lead time of $L$ shifts: production placed on a ghost machine in shift $s$ becomes available in stock in shift $s+L$, captured through $AvailProd_{p,s}$ in the stock balances (A3--A6) and the confirmed-order constraint (C1). Its cost $OutsourcingCost_m$ is charged in the objective at the production (order-placement) shift, independent of $L$.

\end{itemize}
