\newcommand{\desForma}[6]{%
  \pgfmathsetmacro\desFv{sqrt(0.75*(#5)*(#5)+0.25*(#4)*(#4))}%
  \pgfmathsetmacro\desFr{0.34*(#4)}%
  \pgfmathsetmacro\desFg{0.866*((#5)*0.8)}%
  \pgfmathsetmacro\desFt{0.866*((#5)+0.12)}%
  \pgfmathsetmacro\desFs{0.866*((#5)+\desPendente)}%
  \begin{scope}[shift={(#1,#2,#3)},x={(1cm,0cm)},y={(0cm,1cm)}]
    \draw[haste] (0,\desFt) -- (0,\desFs);
    \fill[#6,draw=#6!55!black,line width=0.3pt] (-\desFr,\desFg) rectangle (\desFr,\desFt);
    \fill[#6!65!white,draw=#6!55!black,line width=0.25pt] (0,\desFt) ellipse[x radius=\desFr,y radius=0.5*\desFr];
    \fill[#6,draw=#6!55!black,line width=0.3pt] (0,0) ellipse[x radius=#4,y radius=\desFv];
    \fill[white,opacity=0.35] (-0.38*#4,0.36*\desFv) ellipse[x radius=0.26*#4,y radius=0.2*\desFv];
  \end{scope}%
}

\newcommand{\desProcesso}[1]{%
  \pgfmathtruncatemacro\desK{\pDir==1 ? #1 : 34-#1}%
}

\newcommand{\desMergulho}[1]{%
  \desProcesso{#1}%
  \pgfmathtruncatemacro\desMerg{(\desK>2.5)*(\desK<4.5)+(\desK>5.5)*(\desK<9.5)+(\desK>10.5)*(\desK<12.5)+(\desK>21.5)*(\desK<23.5)}%
}

\newcommand{\desFormasPista}[2]{%
  \foreach \desM in {#1,...,#2} {
    \desMergulho{\desM}
    \pgfmathtruncatemacro\desVest{(\desK>5.5)*(\desK<31.5)}
    \pgfmathtruncatemacro\desComAnel{(\desK>27.5)*(\desK<31.5)}
    \pgfmathsetmacro\desZc{(\desMerg==1 ? \desHb : \desH)-\desPendente-\pC}
    \ifnum\desVest=1
      \desForma{0.5*\desM}{\pY}{\desZc}{\pA}{\pC}{\pCor}
      \ifnum\desComAnel=1
        \desAnel{0.5*\desM}{\pY}{\desZc}{\pA}{\pC}
      \fi
    \else
      \desForma{0.5*\desM}{\pY}{\desZc}{\pA}{\pC}{formanua}
    \fi
  }
}

\newcommand{\desCorrente}[2]{%
  \foreach \desM in {#1,...,#2} {
    \pgfmathtruncatemacro\desMm{\desM+1}
    \desMergulho{\desM}
    \pgfmathsetmacro\desZa{\desMerg==1 ? \desHb : \desH}
    \desMergulho{\desMm}
    \pgfmathsetmacro\desZb{\desMerg==1 ? \desHb : \desH}
    \draw[corrente] (0.5*\desM,\pY,\desZa) -- (0.5*\desM+0.05,\pY,\desZa) -- (0.5*\desM+0.45,\pY,\desZb) -- (0.5*\desM+0.5,\pY,\desZb);
  }
}
