\FloatBarrier

The destination-branch stock balance is applied only to products in $P^{supply}_b$. 
These are products for which Cajamar is responsible for supplying and controlling stock at destination branch $b$. 
The model does not track the full inventory of branches 04 or 07, nor does it model their production processes. 

A product may simultaneously belong to one of Cajamar's product categories, such as 
$P_{intermediate}$, $P_{coil}$, $P_{other}$, $P_{boundary}$, or $P_{06}$, 
and to $P^{supply}_b$ for one or more destination branches. 
The first classification defines its role in Cajamar's stock balance, production structure, purchase logic, and BOM consumption logic. 
The second classification defines whether Cajamar is responsible for supplying and controlling that product as stock at destination branch $b$. 

For each classification, if a product is $P^{supply}_b$, we should add the transfer term and remove $D_{p,1}$. A product should only have demand coming from one branch, for the model purpose.

Demand $D^{supply}_{p,b,w}$ is a preprocessed input obtained by expanding destination-branch downstream demand backward to Cajamar-supplied products, net of relevant downstream stock at the destination branch.

Example: 
06 Demand = 1000
06 Stock = 200

- 06 expandable demand = 800
Exploding factor = 0.5

- 03 Demand = 400

ConfirmedOrders$^{supply}_{p,b,w}$ represents confirmed customer orders assigned to destination branch $b$ for products under Cajamar supply responsibility.

\subsubsection{Objective Function}

\begin{equation}
\begin{aligned}
\min \quad & \alpha \cdot \Biggl[
    \Big(\sum_{p \in P_{finished}} \sum_{w \in W} \!\!  zco_{p,w}  + \\ 
    & \sum_{(p,b,w) \in PBW^{supply}} zsco_{p,b,w} \Big) \cdot PenalizationConfirmedOrders_p\\
    \\
    & + \Big( \sum_{p \in P_{finished}} \sum_{w \in W}  \!\!  zsp_{p,w}  + \\
    & \sum_{(p,b,w) \in PBW^{supply}} zsd_{p,b,w} \Big) \cdot PenalizationDemand_p \\
    \\
    & + \sum_{(p,b,w) \in PBW^{supply}} PenalizationRepositionPoint_{p,b} \cdot zrp_{p,b,w} \\
    & + \sum_{p \in P_{06}} \sum_{w \in W}  \!\! PenalizationSafetyStock_p \cdot zss_{p,w} \\
    & + \sum_{p \in P_{06}} \sum_{w \in W}  \!\! PenalizationCycleStock_p \cdot zcs_{p,w} \\
    & + \sum_{mg \in MG} \sum_{w \in W}  \!\! PenalizationCycleStock_{mg} \cdot zcs^{macro}_{mg,w} \Biggr] \\
    & + \beta \cdot \Biggl[ \sum_{p \in P_{coil}} \sum_{w \in W} CostPurchase_{p,w} \cdot q_{p,w} \Biggr] \\
    & + \gamma \cdot \Biggl[
    \sum_{(p,w) \in PW} \frac{WACC}{52} \cdot UnitCost_p \cdot 
    \frac{ss_{p,w-1} + ss_{p,w}}{2}
    \Biggr] \\    & + \delta \cdot \Biggl[\sum_{p \in P} \sum_{r \in R_p} \sum_{m \in M_r} \sum_{w \in W} CostSetup_p \cdot AvgSetupTime_{p,r,m} \cdot k_{p,r,w}
    \Biggr]\\
    & + \epsilon \cdot \Biggl[
    \sum_{p \in P} \sum_{r \in R_p \cap R_{out}} 
    \sum_{w \in W} CostOutsourcing_{p,r} \cdot x_{p,w,r}
    \Biggr]
\end{aligned}
\label{eq:of}
\end{equation}

\subsubsection{Constraints}

\text{---------------------------------------------------------------------------------------------------------}

\subsubsection*{A1--A2. Stock Balance of Intermediate Products}

\begin{equation}
ss_{p,w} = ss_{p,w-1} - \sum_{p_1 \in P} y_{p_1,p,w} + \sum_{r \in R_p} x_{p,w,r} 
\quad \forall p \in P_{intermediate},\ w = 2,\ldots,32
\label{eq:A1}
\end{equation}

\begin{equation}
ss_{p,1} = StockStart_p - \sum_{p_1 \in P} y_{p_1,p,1} + \sum_{r \in R_p} x_{p,1,r}
\quad \forall p \in P_{intermediate}
\label{eq:A2}
\end{equation}


\noindent\textbf{A3\_1. Cycle stock definition: finished products \& lito products (soft constraint)}
\begin{equation}
ss_{p,w} + Multiplier_{p_{03}(p)} \cdot ss_{p_{03}(p),w}
\geq CycleStock_{p,w} - zcs_{p,w}
\quad \forall p \in P_{06},\ \forall w \in W
\label{eq:A3_1}
\end{equation}



\noindent\textbf{A3\_2. Safety stock definition: finished products \& lito products (soft constraint)}
\begin{equation}
\begin{gathered}
ss_{p,w} + Multiplier_{p_{03}(p)} \cdot ss_{p_{03}(p),w} 
- CycleStock_{p,w} + zcs_{p,w} \\
\geq SafetyStock_{p,w} - zss_{p,w}
\quad \forall p \in P_{06},\ \forall w \in W
\end{gathered}
\label{eq:A3_2}
\end{equation}



\noindent\textbf{A3\_3. Macrogroup cycle stock constraint}
\begin{equation}
\sum_{p \in P(mg)} ss_{p,w} \geq CycleStock^{macro}_{mg,w} 
- zcsmg_{mg,w}
\quad \forall mg \in MG,\ \forall w \in W
\label{eq:A3_5}
\end{equation}



\noindent\textbf{A3\_4. Group-level cycle stock constraint}
\begin{equation}
\begin{gathered}
\sum_{p \in Services(p_{03})} ss_{p,w} + 
Multiplier_{p_{03}} \cdot ss_{p_{03},w}
\geq \sum_{p \in Services(p_{03})} 
\left( CycleStock_{p,w} - zcs_{p,w} \right) \\
\quad \forall p_{03} \in P_{03},\ \forall w \in W
\end{gathered}
\label{eq:A3_3}
\end{equation}

\noindent\textbf{A3\_5. Group-level safety stock constraint}
\begin{equation}
\begin{gathered}
\sum_{p \in Services(p_{03})} ss_{p,w}
+ Multiplier_{p_{03}} \cdot ss_{p_{03},w} \\
\geq \sum_{p \in Services(p_{03})}
\left(
CycleStock_{p,w} - zcs_{p,w}
+ SafetyStock_{p,w} - zss_{p,w}
\right) \\
\forall p_{03} \in P_{03},\ \forall w \in W
\end{gathered}
\label{eq:A3_4}
\end{equation}

\subsubsection{Assumptions:}

\begin{enumerate}
    \item \textbf{Demand is deterministic.} All demand inputs --- confirmed orders, predicted demand, and safety stock and cycle stock targets --- are treated as known and fixed for the duration of each model run. These values are provided by the demand planning module prior to each weekly execution and are not updated within a run.

    \item \textbf{Setup times and unit production times are deterministic.} Both parameters are introduced in the web application (WebApp), as described in Section~\ref{sec:input_data}, and treated as fixed inputs. Variability in operator performance, machine condition, and product-specific complexity is implicitly captured through the historical estimation process.

    \item \textbf{Setups are sequence-independent.} A setup is activated whenever a product is produced on a given routing in a given week, regardless of what was produced on that routing in the previous week. Sequence-dependent changeover times are not modelled, as the weekly granularity of the model makes detailed sequencing decisions the responsibility of the short-term planning module.

    \item \textbf{The shift calendar is a fixed input.} The number of available working days per shift per machine per week is defined by the company's operational calendar and provided as 
    a parameter. The model decides how much of that available capacity to utilise, but does not determine the shift schedule itself.

    \item \textbf{Operators are fully interchangeable within the same production area.} Workers are not assigned to specific machines but to one of four crew groups: lithography, varnishing, stamping, and assembly. All operators within a group are assumed to be equally capable of operating any machine in that group.

    \item \textbf{The availability of non-steel inputs is modelled up to the earliest possible replenishment date, and assumed unlimited thereafter.} For non-steel raw materials and consumables --- such as varnishes and inks --- the model tracks stock levels and scheduled arrivals only up to the week in which a purchase placed at the current moment could first arrive, given the procurement lead time. Beyond that week, availability is assumed to be unlimited, as procurement decisions for these materials fall outside the scope of the model. This ensures that the capacity constraints imposed by current stock levels are respected in the near term, while avoiding the need to model procurement decisions for materials outside the optimisation scope. Only steel coil purchases are explicitly optimised across the full planning horizon.

    \item \textbf{Backordering is allowed, subject to penalisation.} Unmet demand --- whether confirmed orders, predicted demand, or stock target violations --- is penalised in the objective function rather than strictly forbidden. This allows the solver to find feasible solutions under tight capacity conditions. Confirmed orders carry the highest penalty weight, making their fulfilment the model's primary priority.

    \item \textbf{Warehouse capacity is treated as a hard constraint.} The available capacity of each warehouse is adjusted in the data preparation stage to account for stock of products outside the model scope, as described in Section~\ref{sec:data_preparation}. The remaining capacity is enforced as a strict upper bound on the total stock held in each warehouse in each week.

    \item \textbf{The scrap rate is deterministic and estimated per parent product-component pair.} The average quantity of component scrapped per unit of parent product produced is computed from historical production records and incorporated into the BOM consumption factor as a fixed additive term.

    \item \textbf{The model is scoped exclusively to Factory 01.} Production operations at Factory 02 are treated as external events. Finished products that require further processing at Factory 02 are represented in the model through their last intermediate product produced at Factory 01, with demand signals shifted back in time by the corresponding inter-factory lead time, as described in Section~\ref{sec:data_preparation}.

    \item \textbf{Minimum lot sizes are enforced per product-routing combination.} A binary setup variable ensures that production only occurs in quantities at or above the defined minimum lot size, preventing the model from planning economically meaningless micro-batches.

    \item \textbf{Destination branches are represented only through products under Cajamar supply responsibility.}
    The model does not control the full inventory of branches 04 and 07. It only tracks stock for products in $P^{supply}_b$, which are products supplied by Cajamar and required either by the destination branch's operation or by customer demand served from that branch.
    
    \item \textbf{Destination-branch supply demand is preprocessed.}
     The demand $D^{supply}_{p,b,w}$ is calculated before the optimization. It is obtained by expanding destination-branch downstream demand backward to Cajamar-supplied products, after netting available downstream stock at the destination branch and applying the relevant multipliers, and scrap factors.
    
    \item \textbf{Destination-branch confirmed orders are explicitly protected.}
    Confirmed orders assigned to branches 04 or 07 for products in $P^{supply}_b$ are included through $ConfirmedOrders^{supply}_{p,b,w}$ and penalised through $zsco_{p,b,w}$. This allows the model to prioritise confirmed customer orders served from destination branches in the same way it prioritises confirmed orders served from Cajamar.
    
    \item \textbf{Production is planned only at Cajamar.}
    Branches 04 and 07 may have their own production processes, but these are outside the scope of the model. The model only decides how much Cajamar should produce, stock, and transfer to support those branches.
    
    \item \textbf{Transfers from Cajamar to destination branches are available within the same week.}
    At the weekly planning granularity, the transfer time from Cajamar to branches 04 and 07 is assumed to be less than one week. Therefore, a transfer decision $tr_{p,b,w}$ decreases Cajamar stock and increases destination-branch stock in the same week.
    
    \item \textbf{A product may have multiple modelling roles.}
    A product may belong to any Cajamar product category and simultaneously belong to $P^{supply}_b$ for one or more destination branches. The Cajamar product category determines its production, purchase, consumption, and stock balance in Cajamar, while membership in $P^{supply}_b$ determines whether destination stock, destination demand, confirmed orders, and transfers are also modelled - to review how to apply in Pyomo these distinctions!!
    


    
\end{enumerate}
