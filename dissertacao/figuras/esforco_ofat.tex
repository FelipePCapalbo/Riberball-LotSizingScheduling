\newcommand{\painelesforco}[3]{%
  \nextgroupplot[title={#2}, xlabel={#3}]
  \draw[corfrioa, line width=0.5pt, dash pattern=on 2pt off 1.5pt] ({rel axis cs:0,0}|-{axis cs:0,300}) -- ({rel axis cs:1,0}|-{axis cs:0,300});
  \draw[corproposto, line width=0.5pt, dash pattern=on 2pt off 1.5pt] ({rel axis cs:0,0}|-{axis cs:0,600}) -- ({rel axis cs:1,0}|-{axis cs:0,600});
  \addplot[serie fria a] table[x=nivel, y=semanal_segundos]{dados/esforco/ofat_#1.csv};
  \addplot[serie proposta] table[x=nivel, y=diario_segundos]{dados/esforco/ofat_#1.csv};
  \addplot[only marks, mark=*, mark size=2pt, correal] table[x=nivel, y=diario_segundos, restrict expr to domain={\thisrow{base}}{1:1}]{dados/esforco/ofat_#1.csv};
}%
